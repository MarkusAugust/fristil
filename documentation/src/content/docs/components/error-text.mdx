`fs-error-text` er en pure-css tekstkomponent for valideringsfeil og varsler.

## Import

```js
import "@fristil/designsystem/tokens.css"
import "@fristil/designsystem/error-text.css"
```

## TypeScript-API

```ts
import {
  getErrorTextStyleAttributes,
  type ErrorTextVariant,
} from "@fristil/designsystem/error-text"

const variant: ErrorTextVariant = "warning"
const fristilAttributes = getErrorTextStyleAttributes(variant)

// fristilAttributes blir:
// { class: "fs-error-text", "data-variant": "warning" }
```

## Runtime input (for eksempel fra API)

```ts
import { isErrorTextVariant, getErrorTextStyleAttributes } from "@fristil/designsystem/error-text"

const incoming = String(payload.variant)
const variant = isErrorTextVariant(incoming) ? incoming : "error"
const fristilAttributes = getErrorTextStyleAttributes(variant)
```
