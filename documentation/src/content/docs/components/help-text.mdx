`ds-help-text` er en pure-css tekstkomponent for forklaring og kontekst under felter.

## Import

```js
import "@fristil/designsystem/tokens.css"
import "@fristil/designsystem/help-text.css"
```

## TypeScript-API

```ts
import {
  getHelpTextStyleAttributes,
  type HelpTextVariant,
} from "@fristil/designsystem/help-text"

const variant: HelpTextVariant = "success"
const fristilAttributes = getHelpTextStyleAttributes(variant)

// fristilAttributes blir:
// { class: "ds-help-text", "data-variant": "success" }
```

## Runtime input (for eksempel fra API)

```ts
import { isHelpTextVariant, getHelpTextStyleAttributes } from "@fristil/designsystem/help-text"

const incoming = String(payload.variant)
const variant = isHelpTextVariant(incoming) ? incoming : "muted"
const fristilAttributes = getHelpTextStyleAttributes(variant)
```
