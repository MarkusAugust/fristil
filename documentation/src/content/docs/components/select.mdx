`fs-select` er en pure-css stil for `<select>` med samme state-mønster som input.

## Import

```js
import "@fristil/designsystem/tokens.css"
import "@fristil/designsystem/select.css"
```

## TypeScript-API

```ts
import {
  getSelectStyleAttributes,
  type SelectState,
} from "@fristil/designsystem/select"

const state: SelectState = "invalid"
const fristilAttributes = getSelectStyleAttributes(state)

// fristilAttributes blir:
// { class: "fs-select", "data-state": "invalid" }
```

## Runtime input (for eksempel fra API)

```ts
import { isSelectState, getSelectStyleAttributes } from "@fristil/designsystem/select"

const incoming = String(payload.state)
const state = isSelectState(incoming) ? incoming : "default"
const fristilAttributes = getSelectStyleAttributes(state)
```
